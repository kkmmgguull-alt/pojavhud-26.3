          import zlib, struct, os
          os.makedirs("src/main/resources/assets/moin", exist_ok=True)
          w, h = 128, 128
          rows = []
          for y in range(h):
              row = bytearray([0])
              for x in range(w):
                  if x < 8 or x >= 120 or y < 8 or y >= 120:
                      row.extend([92, 58, 33, 255])
                  elif (y // 38) % 2 == 1 and 26 <= (y % 38) <= 34:
                      row.extend([110, 72, 42, 255])
                  else:
                      book = (x // 16) % 4
                      colors = [(185, 55, 55), (55, 130, 185), (65, 165, 85), (210, 165, 45)]
                      r, g, b = colors[book]
                      if x % 16 in (0, 15):
                          r, g, b = 55, 38, 24
                      row.extend([r, g, b, 255])
              rows.append(bytes(row))
          raw = b"".join(rows)
          def chunk(tag, data):
              return struct.pack("!I", len(data)) + tag + data + struct.pack("!I", zlib.crc32(tag + data) & 0xffffffff)
          png = b"\x89PNG\r\n\x1a\n" + chunk(b"IHDR", struct.pack("!IIBBBBB", w, h, 8, 6, 0, 0, 0)) + chunk(b"IDAT", zlib.compress(raw)) + chunk(b"IEND", b"")
          with open("src/main/resources/assets/moin/icon.png", "wb") as f:

          package com.moin;

          import net.minecraft.client.Minecraft;
          import net.minecraft.client.resources.sounds.SimpleSoundInstance;
          import net.minecraft.core.Holder;
          import net.minecraft.sounds.SoundEvent;
          import net.minecraft.sounds.SoundEvents;

          public class SoundHelper {
              public static void playLoad() {
                  play(SoundEvents.PLAYER_LEVELUP, 1.05f, 0.85f);
              }
              public static void playSave() {
                  play(SoundEvents.ENCHANTMENT_TABLE_USE, 1.15f, 0.90f);
              }
              public static void playPreview() {
                  play(SoundEvents.BOOK_PAGE_TURN, 1.05f, 0.95f);
              }
              public static void playDrawerOpen() {
                  play(SoundEvents.CHEST_OPEN, 1.15f, 0.65f);
              }
              public static void playDrawerClose() {
                  play(SoundEvents.CHEST_CLOSE, 1.15f, 0.65f);
              }
              public static void playSelect() {
                  play(SoundEvents.UI_BUTTON_CLICK, 1.10f, 0.75f);
              }
              public static void playDelete() {
                  play(SoundEvents.ITEM_BREAK, 0.90f, 0.80f);
              }
              public static void playError() {
                  play(SoundEvents.NOTE_BLOCK_BASS, 0.75f, 0.85f);
              }

              private static void play(Object soundObj, float pitch, float volume) {
                  try {
                      Minecraft mc = Minecraft.getInstance();
                      if (mc == null || mc.getSoundManager() == null || soundObj == null) return;
                      SoundEvent event = null;
                      if (soundObj instanceof SoundEvent se) {
                          event = se;
                      } else if (soundObj instanceof Holder<?> holder && holder.value() instanceof SoundEvent se) {
                          event = se;
                      }
                      if (event != null) {
                          mc.getSoundManager().play(SimpleSoundInstance.forUI(event, pitch, volume));
                      }
                  } catch (Throwable ignored) {}
              }
          }

                          if (client.gui.screen() == null) {

                      }
                  });

                      return this;

              public static void load() {
                  if (Files.exists(CONFIG_PATH)) {
                      try {
                          String json = Files.readString(CONFIG_PATH);
                          KitFile loaded = GSON.fromJson(json, KitFile.class);
                          if (loaded != null && loaded.kits != null && !loaded.kits.isEmpty()) {
                              data = loaded;
                              return;
                          }
                      } catch (Exception e) {
                          MoinClient.LOGGER.error("Failed to read moin_kits.json", e);
                      }
                  }
                  createDefaultKits();
              }

              public static void save() {
                  try {
                      if (CONFIG_PATH.getParent() != null) {
                          Files.createDirectories(CONFIG_PATH.getParent());
                      }
                      Files.writeString(CONFIG_PATH, GSON.toJson(data));
                  } catch (IOException e) {
                      MoinClient.LOGGER.error("Failed to write moin_kits.json", e);
                  }
              }

              public static String nextAutoKitName() {
                  int idx = 1;
                  while (data.kits.containsKey("kit_" + idx)) {
                      idx++;
                  }

              public static ItemStack buildStack(SlotItem s, RegistryAccess registryAccess) {
                  if (s == null || s.itemId == null || s.itemId.isEmpty()) return ItemStack.EMPTY;
                  Identifier id = Identifier.tryParse(s.itemId.contains(":") ? s.itemId : "minecraft:" + s.itemId);
                  if (id == null) return ItemStack.EMPTY;

                  Item item = BuiltInRegistries.ITEM.getValue(id);
                  if (item == null || item == Items.AIR) return ItemStack.EMPTY;

                  ItemStack stack = new ItemStack(item, Math.max(1, Math.min(99, s.count)));
                  if (s.damage > 0 && stack.isDamageableItem()) {
                      stack.setDamageValue(s.damage);
                  }
                  if (s.customName != null && !s.customName.isEmpty()) {
                      stack.set(DataComponents.CUSTOM_NAME, Component.literal(s.customName));
                  }
                  if (s.potionId != null && !s.potionId.isEmpty()) {
                      String cleanPotion = s.potionId.replace("minecraft:", "");
                      BuiltInRegistries.POTION.listElements()
                          .filter(h -> h.key().identifier().getPath().equals(cleanPotion))
                          .findFirst()
                          .ifPresent(holder -> stack.set(DataComponents.POTION_CONTENTS, new PotionContents(holder)));
                  }
                  if (s.enchantments != null && !s.enchantments.isEmpty() && registryAccess != null) {
                      try {
                          var enchReg = registryAccess.lookupOrThrow(Registries.ENCHANTMENT);
                          for (Map.Entry<String, Integer> e : s.enchantments.entrySet()) {
                              String path = e.getKey().replace("minecraft:", "");
                              int lvl = e.getValue();
                              enchReg.listElements()
                                  .filter(h -> h.key().identifier().getPath().equals(path))
                                  .findFirst()
                                  .ifPresent(holder -> stack.enchant(holder, lvl));
                          }
                      } catch (Exception ignored) {}
                  }
                  return stack;
              }

              public static String saveKit(Minecraft client, String rawName) {
                  if (client == null || client.player == null) {
                      SoundHelper.playError();
                      return "§cMust be in a world to save a kit!";
                  }
                  String key = (rawName == null || rawName.trim().isEmpty())
                      ? nextAutoKitName()
                      : rawName.trim().toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9_-]", "_");

                  Inventory inv = client.player.getInventory();
                  KitEntry entry = new KitEntry(key);

                  for (int slot = 0; slot <= 40; slot++) {
                      ItemStack stack = inv.getItem(slot);
                      if (stack == null || stack.isEmpty()) continue;

                      Identifier itemId = BuiltInRegistries.ITEM.getKey(stack.getItem());
                      SlotItem sItem = new SlotItem(slot, itemId.toString(), stack.getCount());
                      if (stack.isDamaged()) {
                          sItem.damage = stack.getDamageValue();
                      }
                      Component customName = stack.get(DataComponents.CUSTOM_NAME);
                      if (customName != null) {
                          sItem.customName = customName.getString();
                      }
                      PotionContents pot = stack.get(DataComponents.POTION_CONTENTS);
                      if (pot != null && pot.potion().isPresent()) {
                          pot.potion().get().unwrapKey().ifPresent(k -> sItem.potionId = k.identifier().getPath());
                      }
                      ItemEnchantments enchs = stack.getEnchantments();
                      if (enchs != null) {
                          for (Holder<Enchantment> h : enchs.keySet()) {
                              int lvl = enchs.getLevel(h);
                              h.unwrapKey().ifPresent(k -> sItem.enchantments.put(k.identifier().getPath(), lvl));
                          }
                      }
                      entry.items.add(sItem);
                  }

                  data.kits.put(key, entry);
                  save();
                  SoundHelper.playSave();
                  return "§aSaved kit '§e" + key + "§a' (" + entry.items.size() + " items) to moin_kits.json!";
              }

              public static String loadKit(Minecraft client, String rawName) {
                  if (client == null || client.player == null || client.level == null) {
                      SoundHelper.playError();
                      return "§cMust be in a world to load a kit!";
                  }
                  String key = rawName == null ? "" : rawName.trim().toLowerCase(Locale.ROOT);
                  KitEntry kit = data.kits.get(key);
                  if (kit == null) {
                      SoundHelper.playError();
                      return "§cKit '" + key + "' not found! Available: " + String.join(", ", data.kits.keySet());
                  }

                  MinecraftServer server = client.getSingleplayerServer();
                  if (server != null) {
                      UUID uuid = client.player.getUUID();
                      server.execute(() -> {
                          ServerPlayer sp = server.getPlayerList().getPlayer(uuid);
                          if (sp != null) {
                              Inventory sInv = sp.getInventory();
                              for (int i = 0; i <= 40; i++) {
                                  sInv.setItem(i, ItemStack.EMPTY);
                              }
                              RegistryAccess reg = sp.registryAccess();
                              for (SlotItem item : kit.items) {
                                  if (item.slot >= 0 && item.slot <= 40) {
                                      sInv.setItem(item.slot, buildStack(item, reg));
                                  }
                              }
                              sp.inventoryMenu.broadcastChanges();
                          }
                      });
                      SoundHelper.playLoad();
                      return "§aCleared inventory & loaded kit '§e" + key + "§a'!";
                  }

                  if (client.getConnection() != null) {
                      client.getConnection().sendCommand("clear @s");
                      for (SlotItem item : kit.items) {
                          String slotTarget = toMinecraftSlotName(item.slot);
                          if (slotTarget == null) continue;
                          String itemSpec = buildGiveSpec(item);
                          client.getConnection().sendCommand("item replace entity @s " + slotTarget + " with " + itemSpec + " " + item.count);
                      }
                      SoundHelper.playLoad();
                      return "§aSent OP clear & equip commands for kit '§e" + key + "§a'!";
                  }

                  SoundHelper.playError();
                  return "§cUnable to load kit: not connected to a server.";
              }

              private static String toMinecraftSlotName(int slot) {
                  if (slot >= 0 && slot <= 8) return "hotbar." + slot;
                  if (slot >= 9 && slot <= 35) return "inventory." + (slot - 9);
                  return switch (slot) {
                      case 36 -> "armor.feet";
                      case 37 -> "armor.legs";
                      case 38 -> "armor.chest";
                      case 39 -> "armor.head";
                      case 40 -> "weapon.offhand";
                      default -> null;
                  };
              }

              private static String buildGiveSpec(SlotItem item) {
                  String baseId = item.itemId.contains(":") ? item.itemId : "minecraft:" + item.itemId;
                  List<String> comps = new ArrayList<>();
                  if (item.enchantments != null && !item.enchantments.isEmpty()) {
                      List<String> enchParts = new ArrayList<>();
                      for (Map.Entry<String, Integer> e : item.enchantments.entrySet()) {
                          String eid = e.getKey().contains(":") ? e.getKey() : "minecraft:" + e.getKey();
                          enchParts.add("\"" + eid + "\":" + e.getValue());
                      }
                      comps.add("minecraft:enchantments={" + String.join(",", enchParts) + "}");
                  }
                  if (item.potionId != null && !item.potionId.isEmpty()) {
                      String pid = item.potionId.contains(":") ? item.potionId : "minecraft:" + item.potionId;
                      comps.add("minecraft:potion_contents={potion:\"" + pid + "\"}");
                  }
                  if (item.customName != null && !item.customName.isEmpty()) {
                      String escaped = item.customName.replace("\"", "\\\"");
                      comps.add("minecraft:custom_name='\"" + escaped + "\"'");
                  }
                  return comps.isEmpty() ? baseId : baseId + "[" + String.join(",", comps) + "]";
              }
          }

              public static String nextAutoKitName() {
                  int idx = 1;
                  while (data.kits.containsKey("kit_" + idx)) {

              public static ItemStack buildStack(SlotItem s, DynamicRegistryManager registryManager) {
                  if (s == null || s.itemId == null || s.itemId.isEmpty()) return ItemStack.EMPTY;
                  Identifier id = Identifier.tryParse(s.itemId.contains(":") ? s.itemId : "minecraft:" + s.itemId);
                  if (id == null) return ItemStack.EMPTY;

                  Item item = Registries.ITEM.get(id);
                  if (item == Items.AIR) return ItemStack.EMPTY;

                  ItemStack stack = new ItemStack(item, Math.max(1, Math.min(99, s.count)));
                  if (s.damage > 0 && stack.isDamageable()) {
                      stack.setDamage(s.damage);
                  }
                  if (s.customName != null && !s.customName.isEmpty()) {
                      stack.set(DataComponentTypes.CUSTOM_NAME, Text.literal(s.customName));
                  }
                  if (s.potionId != null && !s.potionId.isEmpty()) {
                      String cleanPotion = s.potionId.replace("minecraft:", "");
                      Optional<RegistryEntry.Reference<Potion>> potEntry = Registries.POTION.streamEntries()
                          .filter(e -> e.registryKey().getValue().getPath().equals(cleanPotion))
                          .findFirst();
                      potEntry.ifPresent(entry -> stack.set(DataComponentTypes.POTION_CONTENTS, new PotionContentsComponent(entry)));
                  }
                  if (s.enchantments != null && !s.enchantments.isEmpty() && registryManager != null) {
                      try {
                          var enchReg = registryManager.getOrThrow(RegistryKeys.ENCHANTMENT);
                          for (Map.Entry<String, Integer> e : s.enchantments.entrySet()) {
                              String path = e.getKey().replace("minecraft:", "");
                              int lvl = e.getValue();
                              enchReg.streamEntries()
                                  .filter(h -> h.registryKey().getValue().getPath().equals(path))
                                  .findFirst()
                                  .ifPresent(holder -> stack.addEnchantment(holder, lvl));
                          }
                      } catch (Exception ignored) {}
                  }
                  return stack;
              }

              public static String saveKit(MinecraftClient client, String rawName) {
                  if (client == null || client.player == null) {
                      SoundHelper.playError();
                      return "§cMust be in a world to save a kit!";
                  }
                  String key = (rawName == null || rawName.trim().isEmpty())
                      ? nextAutoKitName()
                      : rawName.trim().toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9_-]", "_");

                  PlayerInventory inv = client.player.getInventory();
                  KitEntry entry = new KitEntry(key);

                  for (int slot = 0; slot <= 40; slot++) {
                      ItemStack stack = inv.getStack(slot);
                      if (stack == null || stack.isEmpty()) continue;

                      Identifier itemId = Registries.ITEM.getId(stack.getItem());
                      SlotItem sItem = new SlotItem(slot, itemId.toString(), stack.getCount());
                      if (stack.isDamaged()) {
                          sItem.damage = stack.getDamage();
                      }
                      Text customName = stack.getCustomName();
                      if (customName != null) {
                          sItem.customName = customName.getString();
                      }
                      PotionContentsComponent pot = stack.get(DataComponentTypes.POTION_CONTENTS);
                      if (pot != null && pot.potion().isPresent()) {
                          pot.potion().get().getKey().ifPresent(k -> sItem.potionId = k.getValue().getPath());
                      }
                      ItemEnchantmentsComponent enchs = stack.getEnchantments();
                      if (enchs != null) {
                          for (RegistryEntry<Enchantment> h : enchs.getEnchantments()) {
                              int lvl = enchs.getLevel(h);
                              h.getKey().ifPresent(k -> sItem.enchantments.put(k.getValue().getPath(), lvl));
                          }
                      }
                      entry.items.add(sItem);
                  }

                  data.kits.put(key, entry);
                  save();
                  SoundHelper.playSave();
                  return "§aSaved kit '§e" + key + "§a' (" + entry.items.size() + " items) to moin_kits.json!";
              }

              public static String loadKit(MinecraftClient client, String rawName) {
                  if (client == null || client.player == null || client.world == null) {
                      SoundHelper.playError();
                      return "§cMust be in a world to load a kit!";
                  }
                  String key = rawName == null ? "" : rawName.trim().toLowerCase(Locale.ROOT);
                  KitEntry kit = data.kits.get(key);
                  if (kit == null) {
                      SoundHelper.playError();
                      return "§cKit '" + key + "' not found! Available: " + String.join(", ", data.kits.keySet());
                  }

                  MinecraftServer server = client.getServer();
                  if (server != null) {
                      UUID uuid = client.player.getUuid();
                      server.execute(() -> {
                          ServerPlayerEntity sp = server.getPlayerManager().getPlayer(uuid);
                          if (sp != null) {
                              PlayerInventory sInv = sp.getInventory();
                              for (int i = 0; i <= 40; i++) {
                                  sInv.setStack(i, ItemStack.EMPTY);
                              }
                              DynamicRegistryManager reg = sp.getRegistryManager();
                              for (SlotItem item : kit.items) {
                                  if (item.slot >= 0 && item.slot <= 40) {
                                      sInv.setStack(item.slot, buildStack(item, reg));
                                  }
                              }
                              sp.playerScreenHandler.sendContentUpdates();
                          }
                      });
                      SoundHelper.playLoad();
                      return "§aCleared inventory & loaded kit '§e" + key + "§a'!";
                  }

                  if (client.player.networkHandler != null) {
                      client.player.networkHandler.sendChatCommand("clear @s");
                      for (SlotItem item : kit.items) {
                          String slotTarget = toMinecraftSlotName(item.slot);
                          if (slotTarget == null) continue;
                          String itemSpec = buildGiveSpec(item);
                          client.player.networkHandler.sendChatCommand("item replace entity @s " + slotTarget + " with " + itemSpec + " " + item.count);
                      }
                      SoundHelper.playLoad();
                      return "§aSent OP clear & equip commands for kit '§e" + key + "§a'!";
                  }

                  SoundHelper.playError();
                  return "§cUnable to load kit: not connected to a server.";
              }

              private static String toMinecraftSlotName(int slot) {
                  if (slot >= 0 && slot <= 8) return "hotbar." + slot;
                  if (slot >= 9 && slot <= 35) return "inventory." + (slot - 9);
                  return switch (slot) {
                      case 36 -> "armor.feet";
                      case 37 -> "armor.legs";
                      case 38 -> "armor.chest";
                      case 39 -> "armor.head";
                      case 40 -> "weapon.offhand";
                      default -> null;
                  };
              }

              private static String buildGiveSpec(SlotItem item) {
                  String baseId = item.itemId.contains(":") ? item.itemId : "minecraft:" + item.itemId;
                  List<String> comps = new ArrayList<>();
                  if (item.enchantments != null && !item.enchantments.isEmpty()) {
                      List<String> enchParts = new ArrayList<>();
                      for (Map.Entry<String, Integer> e : item.enchantments.entrySet()) {
                          String eid = e.getKey().contains(":") ? e.getKey() : "minecraft:" + e.getKey();
                          enchParts.add("\"" + eid + "\":" + e.getValue());
                      }
                      comps.add("minecraft:enchantments={" + String.join(",", enchParts) + "}");
                  }
                  if (item.potionId != null && !item.potionId.isEmpty()) {
                      String pid = item.potionId.contains(":") ? item.potionId : "minecraft:" + item.potionId;
                      comps.add("minecraft:potion_contents={potion:\"" + pid + "\"}");
                  }
                  if (item.customName != null && !item.customName.isEmpty()) {
                      String escaped = item.customName.replace("\"", "\\\"");
                      comps.add("minecraft:custom_name='\"" + escaped + "\"'");
                  }
                  return comps.isEmpty() ? baseId : baseId + "[" + String.join(",", comps) + "]";
              }
          }

              private ItemStack drawSlot(DrawContext context, int x, int y, int slot, String hint, int mouseX, int mouseY, int bg) {

                  context.fill(x, y, x + 17, y + 17, hover ? 0xFF4A6FA5 : bg);

                      context.drawCenteredTextWithShadow(this.textRenderer, "§8" + hint, x + 9, y + 5, 0x666666);
                  }
                  return ItemStack.EMPTY;
              }
